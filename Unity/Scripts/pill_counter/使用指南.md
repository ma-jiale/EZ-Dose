```
检查: Camera Index（尝试0, 1, 2）
检查: 摄像头权限

### Q2: 计数不准确？
```
检查: 光照是否均匀
检查: 背景是否纯净

```
确保: 摄像头不晃动

操作: 按B手动捕捉
```

### 完整集成示例
```csharp
using EZDose.PillCounter;

public class DispensingWorkflow : MonoBehaviour
{
    public PillCounterController pillCounter;
    public int targetCount = 10;
    
    void Start()
    {
        // 确保背景已捕捉
        StartCoroutine(WaitForBackground());
    }
    
    IEnumerator WaitForBackground()
    {
        while (!pillCounter.IsBackgroundCaptured())
        {
            yield return new WaitForSeconds(0.1f);
        }
        
        Debug.Log("背景已就绪，开始计数流程");
        StartCoroutine(MonitorCount());
    }
    
    IEnumerator MonitorCount()
    {
        while (true)
        {
            int currentCount = pillCounter.GetCurrentPillCount();
            
            if (currentCount == targetCount)
            {
                Debug.Log($"✅ 已达到目标数量: {targetCount}");
                OnTargetReached();
                break;
            }
            else if (currentCount > targetCount)
            {
                Debug.LogWarning($"⚠️ 超出目标: {currentCount}/{targetCount}");
            }
            
            yield return new WaitForSeconds(0.5f);
        }
    }
    
    void OnTargetReached()
    {
        // 执行后续流程...
    }
}
```
